+    def sample(
+        self, 
+        tokenizer: PreTrainedTokenizerBase, 
+        num_requests: int,
+        input_tokens: int = 122880,
+        output_tokens: int = 8192,
+        tool_subset: Optional[list[str]] = None,

+            output_tokens: Expected output token length (default: 8192)

+            **kwargs: Additional arguments
+
